export const ParticleAsk = () => {
  const W = 96;
  const H = 46;
  const GAP = 4;
  const canvasRef = useRef(null);
  const state = useRef({ hover: false, mx: 0, my: 0, hoverStart: 0 });
  const [dark, setDark] = useState(false);
  useEffect(() => {
    const el = document.documentElement;
    const update = () => setDark(el.classList.contains("dark"));
    update();
    const mo = new MutationObserver(update);
    mo.observe(el, { attributes: true, attributeFilter: ["class"] });
    return () => mo.disconnect();
  }, []);
  useEffect(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;
    const ctx = canvas.getContext("2d");
    const dpr = window.devicePixelRatio || 1;
    canvas.width = W * dpr;
    canvas.height = H * dpr;
    ctx.scale(dpr, dpr);
    const reduce = window.matchMedia && window.matchMedia("(prefers-reduced-motion: reduce)").matches;
    let seed = 12345;
    const rand = () => { seed = (seed * 16807) % 2147483647; return seed / 2147483647; };
    const cols = Math.floor((W - 4) / GAP) + 1;
    const rows = Math.floor((H - 4) / GAP) + 1;
    const ox = (W - (cols - 1) * GAP) / 2;
    const oy = (H - (rows - 1) * GAP) / 2;
    const pts = [];
    for (let i = 0; i < cols; i++) {
      for (let j = 0; j < rows; j++) {
        const edge = i === 0 || j === 0 || i === cols - 1 || j === rows - 1;
        const x = ox + i * GAP;
        const y = oy + j * GAP;
        const a = rand() * Math.PI * 2;
        const d = 20 + rand() * 60;
        pts.push({
          x, y, edge,
          sx: x + Math.cos(a) * d, sy: y + Math.sin(a) * d,
          px: rand() * 6.28, py: rand() * 6.28, sp: 0.6 + rand() * 0.8,
          base: edge ? 0.9 : 0.32 + rand() * 0.12,
        });
      }
    }
    const t0 = performance.now();
    let raf = 0;
    const ease = (k) => 1 - Math.pow(1 - k, 3);
    const beat = (t) => {
      const period = 60 / 44;
      const p = (t % period) / period;
      const pulse = (c) => Math.max(0, 1 - Math.abs(p - c) / 0.06);
      return Math.max(pulse(0.08), 0.65 * pulse(0.08 + 0.18));
    };
    const frame = (now) => {
      const t = (now - t0) / 1000;
      const s = state.current;
      const enter = reduce ? 1 : ease(Math.min(1, t / 0.6));
      ctx.fillStyle = dark ? (reduce ? "#F5F5F5" : "rgba(245,245,245,0.45)") : (reduce ? "#000" : "rgba(0,0,0,0.45)");
      ctx.fillRect(0, 0, W, H);
      const ht = (now - s.hoverStart) / 1000;
      for (const p of pts) {
        let x = p.sx + (p.x - p.sx) * enter;
        let y = p.sy + (p.y - p.sy) * enter;
        if (!reduce) {
          x += Math.sin(t * p.sp * 2 + p.px) * 0.35;
          y += Math.cos(t * p.sp * 2 + p.py) * 0.35;
        }
        let r = 0.55;
        let b = p.base;
        if (s.hover && !reduce) {
          const dx = x - s.mx;
          const dy = y - s.my;
          const dist = Math.sqrt(dx * dx + dy * dy) || 1;
          const near = Math.max(0, 1 - dist / 26);
          r += near * 0.6;
          b = Math.min(1, b + near * 0.6);
          x -= (dx / dist) * near * 1.2;
          y -= (dy / dist) * near * 1.2;
          const ring = (ht * 60) % 70;
          const w = Math.max(0, 1 - Math.abs(dist - ring) / 8) * beat(ht);
          r += w * 0.5;
          b = Math.min(1, b + w * 0.5);
          x += (dx / dist) * w * 1.5;
          y += (dy / dist) * w * 1.5;
        }
        ctx.fillStyle = (dark ? "rgba(17,17,17," : "rgba(255,255,255,") + b.toFixed(3) + ")";
        ctx.beginPath();
        ctx.arc(x, y, r, 0, 6.2832);
        ctx.fill();
      }
      if (!reduce) raf = requestAnimationFrame(frame);
    };
    raf = requestAnimationFrame(frame);
    return () => cancelAnimationFrame(raf);
  }, [dark]);
  const onMove = (e) => {
    const rect = e.currentTarget.getBoundingClientRect();
    state.current.mx = ((e.clientX - rect.left) / rect.width) * W;
    state.current.my = ((e.clientY - rect.top) / rect.height) * H;
  };
  const label = { position: "absolute", inset: 0, display: "flex", alignItems: "center", justifyContent: "center", pointerEvents: "none" };
  return (
    <button
      type="submit"
      aria-label="Ask"
      onMouseEnter={(e) => { state.current.hover = true; state.current.hoverStart = performance.now(); onMove(e); }}
      onMouseLeave={() => { state.current.hover = false; }}
      onMouseMove={onMove}
      style={{ position: "relative", width: W, height: H, flex: "none", padding: 0, border: "none", borderRadius: 4, background: dark ? "#F5F5F5" : "#000", cursor: "pointer", overflow: "hidden" }}
    >
      <canvas ref={canvasRef} style={{ position: "absolute", inset: 0, width: W, height: H, display: "block" }} />
      <span style={{ ...label, zIndex: 1, color: dark ? "#F5F5F5" : "#000", WebkitTextStroke: dark ? "3px #F5F5F5" : "3px #000", fontWeight: 600, fontSize: 16 }}>Ask</span>
      <span style={{ ...label, zIndex: 2, color: dark ? "#111111" : "#fff", fontWeight: 600, fontSize: 16 }}>Ask</span>
    </button>
  );
};

export const AskJuo = () => {
  const examples = [
    { label: "Discount not applied", q: "Why wasn't a subscription discount applied on a renewal, and how do I fix it?" },
    { label: "iDEAL and SEPA renewals", q: "How do iDEAL and SEPA subscriptions renew, and why can payment take a few days?" },
    { label: "Failed payment emails", q: "How do I send Klaviyo emails when a subscription payment fails, with a link to update the payment method?" },
  ];
  const [question, setQuestion] = useState("");
  const formRef = useRef(null);
  const ask = (text) => {
    const q = text.trim();
    window.location.href = window.location.pathname + (q ? "?assistant=" + encodeURIComponent(q) : "?assistant");
  };
  const placeDialog = () => {
    const form = formRef.current;
    const dialog = document.querySelector('[data-search-item="dialog"]');
    if (!form || !dialog || window.innerWidth < 768) return;
    const fr = form.getBoundingClientRect();
    const dr = dialog.getBoundingClientRect();
    const cs = getComputedStyle(dialog);
    const top = parseFloat(cs.marginTop) + (fr.top - dr.top) - (dr.height - fr.height) / 2;
    const left = parseFloat(cs.marginLeft) + (fr.left - dr.left);
    dialog.style.margin = top + "px 0 0 " + left + "px";
    dialog.style.width = fr.width + "px";
    dialog.style.maxWidth = fr.width + "px";
  };
  const openSearch = (text) => {
    try {
      const find = () => document.querySelector('[data-search-item="input"]');
      const press = (mod) => document.dispatchEvent(new KeyboardEvent("keydown", { key: "k", code: "KeyK", metaKey: mod === "meta", ctrlKey: mod === "ctrl", bubbles: true, cancelable: true }));
      if (!find()) press("meta");
      let tries = 0;
      const fill = () => {
        const inp = find();
        if (!inp) {
          tries += 1;
          if (tries === 3) press("ctrl");
          if (tries < 20) setTimeout(fill, 30);
          return;
        }
        placeDialog();
        if (text) {
          const setter = Object.getOwnPropertyDescriptor(HTMLInputElement.prototype, "value").set;
          setter.call(inp, text);
          inp.dispatchEvent(new Event("input", { bubbles: true }));
        }
        inp.focus();
      };
      setTimeout(fill, 30);
    } catch (e) {}
  };
  return (
    <div className="not-prose" style={{ maxWidth: 640, margin: "0 auto 56px", display: "grid", gap: 12 }}>
      <style>{`html[data-current-path="/"] #navigation-items > div:has(> .sidebar-group-header),
html[data-current-path="/docs"] #navigation-items > div:has(> .sidebar-group-header),
html[data-current-path="/"] #sidebar-content div:has(> div > #search-bar-entry),
html[data-current-path="/docs"] #sidebar-content div:has(> div > #search-bar-entry),
html[data-current-path="/"] #navigation-items > ul:not(:first-child),
html[data-current-path="/docs"] #navigation-items > ul:not(:first-child),
html[data-current-path="/"] .nav-tabs-item[data-active] > div.absolute,
html[data-current-path="/docs"] .nav-tabs-item[data-active] > div.absolute {
  display: none !important;
}
html[data-current-path="/"] .nav-tabs-item[data-active],
html[data-current-path="/docs"] .nav-tabs-item[data-active] {
  color: rgb(81, 81, 87) !important;
  text-shadow: none !important;
}
html.dark[data-current-path="/"] .nav-tabs-item[data-active],
html.dark[data-current-path="/docs"] .nav-tabs-item[data-active] {
  color: rgb(160, 159, 165) !important;
}
/* Grayscale home styles. Move to style.css (it already has the same rules) and delete from here. */
html { --juo-surface: #fbfbfc; --juo-raised: #f3f3f4; --juo-border: #ebebeb; --juo-border-strong: #d4d4d4; --juo-field-border: #8f8f8f; --juo-field-focus: #555555; --juo-text: #0f1011; --juo-text-2: #6b6f76; --juo-icon-2: #6b6f76; }
html.dark { --juo-surface: #1b1b1b; --juo-raised: #333333; --juo-border: #404040; --juo-border-strong: #555555; --juo-field-border: #6b6b6b; --juo-field-focus: #9e9e9e; --juo-text: #f5f5f5; --juo-text-2: #c4c4c4; --juo-icon-2: #9e9e9e; }
.juo-ask-input { flex: 1; min-width: 0; height: 46px; box-sizing: border-box; padding: 0 14px; border-radius: 4px; border: 1px solid var(--juo-field-border); background: var(--juo-surface); color: var(--juo-text); font: inherit; outline: none; }
.juo-ask-input::placeholder { color: var(--juo-icon-2); }
.juo-ask-input:hover { border-color: var(--juo-field-focus); }
.juo-ask-input:focus { border-color: var(--juo-text); }
.juo-ask-chip { padding: 4px 9px; border-radius: 2px; border: 1px solid var(--juo-border); background: var(--juo-surface); color: var(--juo-text-2); font: inherit; cursor: pointer; white-space: nowrap; }
.juo-ask-chip:hover { background: var(--juo-raised); border-color: var(--juo-border-strong); color: var(--juo-text); }
.juo-tile { border-radius: 4px !important; border-color: var(--juo-border) !important; background: var(--juo-surface) !important; }
.juo-tile:hover { border-color: var(--juo-border-strong) !important; }
.juo-tile-visual { border-bottom-color: var(--juo-border) !important; }
.juo-tile-icon { opacity: 1 !important; color: var(--juo-icon-2) !important; }
.juo-tile-icon svg { background-color: var(--juo-icon-2) !important; }
.juo-tile:hover .juo-tile-icon svg { background-color: var(--juo-text) !important; }
.juo-tile-title { color: var(--juo-text) !important; }
.juo-tile-text { color: var(--juo-text-2) !important; }`}</style>
      <form
        ref={formRef}
        onSubmit={(e) => { e.preventDefault(); ask(question); }}
        style={{ display: "flex", gap: 8 }}
      >
        <input
          type="text"
          aria-label="Search or ask a question"
          value={question}
          onChange={(e) => { const v = e.target.value; setQuestion(""); e.target.blur(); openSearch(v); }}
          onFocus={(e) => { e.target.blur(); openSearch(""); }}
          placeholder="Search or ask a question"
          className="juo-ask-input"
        />
        <ParticleAsk />
      </form>
      <div style={{ display: "flex", gap: 6, flexWrap: "nowrap", justifyContent: "center", fontSize: 12, overflow: "hidden" }}>
        {examples.map((ex) => (
          <button
            key={ex.label}
            type="button"
            onClick={() => ask(ex.q)}
            className="juo-ask-chip"
          >
            {ex.label}
          </button>
        ))}
      </div>
    </div>
  );
};
