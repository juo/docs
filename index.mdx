export const AskJuo = () => {
  const examples = [
    "Set up a cancellation flow",
    "Migrate from another app",
    "Add the subscription widget",
  ];
  const [question, setQuestion] = useState("");
  const ask = (text) => {
    const q = text.trim();
    if (!q) return;
    window.location.href = window.location.pathname + "?assistant=" + encodeURIComponent(q);
  };
  return (
    <div className="not-prose" style={{ maxWidth: 640, margin: "0 auto 56px", display: "grid", gap: 12 }}>
      <style>{`html[data-current-path="/"] #navigation-items > div:has(> .sidebar-group-header),
html[data-current-path="/docs"] #navigation-items > div:has(> .sidebar-group-header),
html[data-current-path="/"] #sidebar-content div:has(> div > #search-bar-entry),
html[data-current-path="/docs"] #sidebar-content div:has(> div > #search-bar-entry) {
  display: none !important;
}`}</style>
      <form
        onSubmit={(e) => { e.preventDefault(); ask(question); }}
        style={{ display: "flex", gap: 8 }}
      >
        <input
          type="text"
          aria-label="Ask a question about Juo"
          value={question}
          onChange={(e) => setQuestion(e.target.value)}
          placeholder="Ask a question about Juo"
          style={{ flex: 1, minWidth: 0, padding: "12px 14px", borderRadius: 10, border: "1px solid rgba(127,127,127,.35)", background: "transparent", color: "inherit", font: "inherit" }}
        />
        <button
          type="submit"
          style={{ padding: "12px 18px", borderRadius: 10, border: "none", background: "#3727F4", color: "#ffffff", font: "inherit", fontWeight: 500, cursor: "pointer" }}
        >
          Ask
        </button>
      </form>
      <div style={{ display: "flex", gap: 8, flexWrap: "wrap", justifyContent: "center", fontSize: 13 }}>
        {examples.map((ex) => (
          <button
            key={ex}
            type="button"
            onClick={() => ask(ex)}
            style={{ padding: "5px 10px", borderRadius: 999, border: "1px solid rgba(127,127,127,.3)", background: "transparent", color: "inherit", font: "inherit", cursor: "pointer", opacity: 0.85, whiteSpace: "nowrap" }}
          >
            {ex}
          </button>
        ))}
      </div>
    </div>
  );
};
