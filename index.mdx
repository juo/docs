export const AskJuo = () => {
  const examples = [
    "How do I set up a cancellation flow?",
    "How do I move subscriptions from another app?",
    "How do I add the subscription widget to my store?",
  ];
  const [question, setQuestion] = useState("");
  const ask = (text) => {
    const q = text.trim();
    if (!q) return;
    window.location.href = window.location.pathname + "?assistant=" + encodeURIComponent(q);
  };
  return (
    <div className="not-prose" style={{ border: "1px solid rgba(127,127,127,.25)", borderRadius: 12, padding: 20, display: "grid", gap: 14, margin: "8px 0 32px" }}>
      <form
        onSubmit={(e) => { e.preventDefault(); ask(question); }}
        style={{ display: "flex", gap: 8, flexWrap: "wrap" }}
      >
        <label style={{ flex: "1 1 260px", minWidth: 0, display: "grid", gap: 6, fontSize: 14 }}>
          Ask a question about Juo
          <input
            type="text"
            value={question}
            onChange={(e) => setQuestion(e.target.value)}
            placeholder="For example: how do I offer a delay instead of a cancellation?"
            style={{ width: "100%", padding: "10px 12px", borderRadius: 8, border: "1px solid rgba(127,127,127,.35)", background: "transparent", color: "inherit", font: "inherit" }}
          />
        </label>
        <button
          type="submit"
          style={{ alignSelf: "end", padding: "10px 16px", borderRadius: 8, border: "none", background: "#3727F4", color: "#ffffff", font: "inherit", fontWeight: 500, cursor: "pointer" }}
        >
          Ask
        </button>
      </form>
      <div style={{ display: "flex", gap: 8, flexWrap: "wrap", fontSize: 13 }}>
        {examples.map((ex) => (
          <button
            key={ex}
            type="button"
            onClick={() => ask(ex)}
            style={{ padding: "6px 10px", borderRadius: 999, border: "1px solid rgba(127,127,127,.3)", background: "transparent", color: "inherit", font: "inherit", cursor: "pointer" }}
          >
            {ex}
          </button>
        ))}
      </div>
    </div>
  );
};
