export const AskJuo = () => {
  const examples = [
    "How do I set up a cancellation flow?",
    "How do I move subscriptions from another app?",
    "How do I add the subscription widget?",
  ];
  const [question, setQuestion] = useState("");
  const ask = (text) => {
    const q = text.trim();
    if (!q) return;
    window.location.href = window.location.pathname + "?assistant=" + encodeURIComponent(q);
  };
  return (
    <div className="not-prose" style={{ maxWidth: 560, margin: "0 auto 48px", display: "grid", gap: 14, textAlign: "center" }}>
      <h1 style={{ fontSize: 28, fontWeight: 560, letterSpacing: "-0.015em", margin: 0 }}>Ask a question about Juo</h1>
      <form
        onSubmit={(e) => { e.preventDefault(); ask(question); }}
        style={{ display: "flex", gap: 8 }}
      >
        <input
          type="text"
          aria-label="Ask a question about Juo"
          value={question}
          onChange={(e) => setQuestion(e.target.value)}
          placeholder="How do I offer a delay instead of a cancellation?"
          style={{ flex: 1, minWidth: 0, padding: "12px 14px", borderRadius: 10, border: "1px solid rgba(127,127,127,.35)", background: "transparent", color: "inherit", font: "inherit" }}
        />
        <button
          type="submit"
          style={{ padding: "12px 18px", borderRadius: 10, border: "none", background: "#3727F4", color: "#ffffff", font: "inherit", fontWeight: 500, cursor: "pointer" }}
        >
          Ask
        </button>
      </form>
      <div style={{ display: "flex", gap: 8, flexWrap: "wrap", justifyContent: "center", fontSize: 13 }}>
        {examples.map((ex) => (
          <button
            key={ex}
            type="button"
            onClick={() => ask(ex)}
            style={{ padding: "6px 10px", borderRadius: 999, border: "1px solid rgba(127,127,127,.3)", background: "transparent", color: "inherit", font: "inherit", cursor: "pointer", opacity: 0.85 }}
          >
            {ex}
          </button>
        ))}
      </div>
    </div>
  );
};
